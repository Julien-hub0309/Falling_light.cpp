cmake_minimum_required(VERSION 3.16)
project(FallingLight LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# raylib : utilise une version installée, sinon la télécharge et la compile automatiquement
find_package(raylib 5.0 QUIET)
if(NOT raylib_FOUND)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(raylib)
endif()

file(GLOB SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp)
add_executable(falling_light ${SOURCES})
target_include_directories(falling_light PRIVATE ${CMAKE_SOURCE_DIR}/src)
target_link_libraries(falling_light PRIVATE raylib)
if(NOT WIN32)
  target_link_libraries(falling_light PRIVATE m)
endif()
if(MSVC)
  target_compile_options(falling_light PRIVATE /utf-8)
endif()
